NAME
     pc - critical pressure

LIBRARY
     iapws library (libiapws, -liapws)

SYNOPSIS
     Fortran   real(dp), parameter :: pc_H2O

               real(dp), parameter :: pc_D2O

     C         const double iapws_r283_pc_H2O;

               const double iapws_r283_pc_D2O;

     Python    float:pc_H2O

               float:pc_D2O

DESCRIPTION
     Critical  pressure  constants for water, .pc_H2O, and heavy water,
     pc_D2O, used for computing the water properties.

EXAMPLES
     Fortran
         use iapws
         real(dp) :: T = pc_H2O
         real(dp) :: T = pc_D2O

     C
         #include "iapws.h"
         double T;
         T = iapws_r283_pc_H2O;
         T = iapws_r283_pc_D2O;

     Python:
         import pyiapws
         T = pc_H2O
         T = pc_D2O
